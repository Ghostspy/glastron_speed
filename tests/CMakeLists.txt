cmake_minimum_required(VERSION 3.16)
project(boat_speed_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME build-source-guards COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_build_modes.py)
add_library(speed-model STATIC ../main/speed_model.c ../main/gps_nmea.c ../main/gps_ublox.c)
target_include_directories(speed-model PUBLIC ../main)
target_compile_options(speed-model PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(speed-model PUBLIC m)
foreach(UNIT model gps ublox)
    add_executable(test-${UNIT} test_${UNIT}.c)
    target_compile_options(test-${UNIT} PRIVATE -Wall -Wextra -Werror -pedantic)
    target_link_libraries(test-${UNIT} PRIVATE speed-model)
    add_test(NAME ${UNIT} COMMAND test-${UNIT})
endforeach()

add_executable(test-uart test_uart.c ../main/gps_uart.c)
target_include_directories(test-uart PRIVATE fake_idf)
target_compile_options(test-uart PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(test-uart PRIVATE speed-model)
add_test(NAME uart-provider COMMAND test-uart)

add_executable(test-uart-passive test_uart.c ../main/gps_uart.c)
target_include_directories(test-uart-passive PRIVATE fake_idf)
target_compile_definitions(test-uart-passive PRIVATE CONFIG_SPEED_GPS_AUTOCONFIG=0)
target_compile_options(test-uart-passive PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(test-uart-passive PRIVATE speed-model)
add_test(NAME uart-passive COMMAND test-uart-passive)

file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS ../components/lvgl/src/*.c)
add_library(lvgl-host STATIC ${LVGL_SOURCES})
target_include_directories(lvgl-host PUBLIC ../components/lvgl ..)
target_compile_definitions(lvgl-host PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_compile_options(lvgl-host PRIVATE -Wno-unused-parameter)
add_executable(render-ui render_ui.c ../main/speed_ui.c ../main/speed_font.c
    ../main/speed_startup.c ../main/glastron_logo.c)
target_include_directories(render-ui PRIVATE ../main)
target_compile_options(render-ui PRIVATE -Wall -Wextra -Werror)
target_link_libraries(render-ui PRIVATE lvgl-host speed-model)

add_executable(test-startup test_startup.c ../main/speed_startup.c ../main/glastron_logo.c)
target_include_directories(test-startup PRIVATE ../main)
target_compile_options(test-startup PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test-startup PRIVATE lvgl-host m)
add_test(NAME startup-timing COMMAND test-startup)

foreach(SCENE normal low empty missing speed-missing fuel-missing gps gps-waiting stopped night startup course-missing course-slow course-north)
    add_test(NAME render-${SCENE} COMMAND render-ui ${SCENE} ${CMAKE_CURRENT_BINARY_DIR}/${SCENE}.ppm)
endforeach()
